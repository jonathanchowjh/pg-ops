# pgops v0.1 Triage Cockpit Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `pgops` (bare) a live first-five-minutes triage screen that drills into `locks`, `top`, and a new `sessions` view (with kill), with `--snapshot` parity everywhere — per `docs/superpowers/specs/2026-07-12-v0.1-triage-cockpit-design.md`.

**Architecture:** Each command keeps the established shape — SQL in `query.ts` → pure `model.ts` transform → pure `render.ts` markdown → Ink component → `index.tsx` runner. Kill is generalized into `src/core/kill.ts` (one hardened path, two callers). Hub navigation is a root `HubApp` in `src/commands/home/index.tsx` holding a one-deep view stack; child views accept an optional `onBack` and behave exactly as today when it's absent.

**Tech Stack:** TypeScript (ESM, `.js` import suffixes), Ink 5 + React 18, `pg`, vitest + ink-testing-library. No new runtime dependencies.

## Global Constraints

- Node >= 20; ESM only; relative imports end in `.js`.
- Runtime deps stay exactly: `ink`, `pg`, `react`. No additions.
- All SQL is static text with `$n` parameters — never string-built.
- The ONLY write action in the codebase is `pg_terminate_backend` inside `src/core/kill.ts`; everything else read-only.
- Colors come from `theme` in `src/core/theme.ts` (`critical`/`warning`/`info`/`success`/`dim`).
- Numeric SQL results (`EXTRACT(EPOCH …)`, `count`, `bigint`) arrive as strings from `pg` — coerce with `Number()` in `query.ts` mappers (see `src/commands/locks/query.ts:78` for the pattern).
- Verdict thresholds (from spec §3.1): idle-in-txn warn > 60 s, critical ≥ 300 s · blocked warn ≥ 1 pid, critical when longest wait ≥ 60 s · long-running warn > 30 s, critical > 300 s · connections warn when headroom < 20 %, critical < 10 % · wraparound warn ≥ 75 %, critical ≥ 90 % · slots warn when any inactive slot retains > 1 GB (1_073_741_824 bytes).
- Run `npm test` and `npm run typecheck` before every commit; both must pass.
- Poll cadence for all live views: `REFRESH_MS = 2_000`.

```typescript
import { query } from './connection.js';
import { writeAudit } from './audit.js';

export type KillOutcome = 'terminated' | 'session-changed' | 'gone';

export interface KillTarget {
  pid: number;
  /** Session identity captured at selection (pg_stat_activity.backend_start::text). */
  backendStart: string | null;
}

export interface KillContext {
  subcommand: string;
  reason: string;
  meta?: Record<string, unknown>;
}

// (carry over the KILL_SQL constant and the full doc comments from
// src/commands/locks/kill.ts — the PID-reuse-guard rationale must move with the code)
const KILL_SQL = `
SELECT CASE WHEN backend_start = $2::timestamptz
            THEN pg_terminate_backend(pid)
            ELSE NULL END AS terminated
FROM pg_stat_activity
WHERE pid = $1
`;

export async function killBackend(target: KillTarget, ctx: KillContext): Promise<KillOutcome> {
  let outcome: KillOutcome = 'gone';
  let result = 'unknown';
  try {
    const rows = await query<{ terminated: boolean | null }>(KILL_SQL, [target.pid, target.backendStart]);
    const row = rows[0];
    if (!row) outcome = 'gone';
    else if (row.terminated === true) outcome = 'terminated';
    else if (row.terminated === null) outcome = 'session-changed';
    else outcome = 'gone';
    result = outcome;
    return outcome;
  } catch (err) {
    result = `error: ${(err as Error).message}`;
    throw err;
  } finally {
    await writeAudit({
      action: 'kill',
      subcommand: ctx.subcommand,
      target: { pid: target.pid, backend_start: target.backendStart, ...ctx.meta },
      reason: ctx.reason,
      result,
    }).catch(() => {
      // Never let an audit-log write failure mask the kill outcome.
    });
  }
}
```

- [ ] **Step 4: Implement `model.ts`**:

```typescript
import type { SessionRow } from './query.js';

export type SessionSort = 'triage' | 'active';

const IDLE_IN_TXN = 'idle in transaction';

function triageRank(r: SessionRow): number {
  if (r.state === IDLE_IN_TXN || r.state === 'idle in transaction (aborted)') return 0;
  if (r.state === 'active') return 1;
  return 2;
}

export function prepareSessions(rows: SessionRow[], opts: { sort: SessionSort; filter: string }): SessionRow[] {
  const needle = opts.filter.trim().toLowerCase();
  const filtered = needle
    ? rows.filter((r) =>
        [r.user, r.app, r.state, r.query].some((f) => f?.toLowerCase().includes(needle)))
    : rows;
  const sorted = [...filtered];
  if (opts.sort === 'triage') {
    sorted.sort((a, b) =>
      triageRank(a) - triageRank(b)
      || (triageRank(a) === 0
        ? (b.inStateSecs ?? 0) - (a.inStateSecs ?? 0)
        : (b.runningSecs ?? 0) - (a.runningSecs ?? 0)));
  } else {
    const rank = (r: SessionRow) => (r.state === 'active' ? 0 : 1);
    sorted.sort((a, b) => rank(a) - rank(b) || (b.runningSecs ?? 0) - (a.runningSecs ?? 0));
  }
  return sorted;
}
```

import type { SessionRow } from './query.js';

export function sessionsToMarkdown(rows: SessionRow[]): string {

import type {
  BlockedPanel, ConnectionsPanel, LongRunningRow, SessionsPanel, SlotsCard, WraparoundCard,
} from './query.js';

export type VerdictLevel = 'ok' | 'warn' | 'critical';
export interface Verdict { level: VerdictLevel; note?: string }

const GB = 1024 ** 3;

export function verdictSessions(p: SessionsPanel): Verdict {
  const s = p.oldestIdleInTxnSecs;
  if (s == null || s <= 60) return { level: 'ok' };
  return { level: s >= 300 ? 'critical' : 'warn', note: `oldest idle-in-transaction ${Math.round(s)}s` };
}

export function verdictBlocked(p: BlockedPanel): Verdict {
  if (p.blockedCount === 0) return { level: 'ok' };
  const critical = (p.longestWaitSecs ?? 0) >= 60;
  return { level: critical ? 'critical' : 'warn', note: `${p.blockedCount} pid(s) waiting` };
}

export function verdictLongRunning(rows: LongRunningRow[]): Verdict {
  const longest = rows[0]?.runningSecs ?? 0;
  if (longest > 300) return { level: 'critical', note: `pid ${rows[0]?.pid} running ${Math.round(longest)}s` };
  if (longest > 30) return { level: 'warn' };
  return { level: 'ok' };
}

export function verdictConnections(p: ConnectionsPanel): Verdict {
  const pct = p.max > 0 ? (p.headroom / p.max) * 100 : 100;
  if (pct < 10) return { level: 'critical', note: `${p.headroom} connections left` };
  if (pct < 20) return { level: 'warn', note: `${p.headroom} connections left` };
  return { level: 'ok' };
}

export function verdictWraparound(c: WraparoundCard): Verdict {
  if (c.pctOfHorizon >= 90) return { level: 'critical', note: 'see dba-playbook/03' };
  if (c.pctOfHorizon >= 75) return { level: 'warn', note: 'see dba-playbook/03' };
  return { level: 'ok' };
}

export function verdictSlots(c: SlotsCard): Verdict {
  if (c.inactiveCount > 0 && c.retainedBytes > GB) {
    return { level: 'warn', note: `inactive slot '${c.worstSlot}' retaining WAL — see dba-playbook/04` };
  }
  return { level: 'ok' };

const SESSIONS_SQL = `
SELECT state,
       count(*)::int                                          AS sessions,
       max(EXTRACT(EPOCH FROM (now() - state_change)))        AS longest_secs
FROM pg_stat_activity
WHERE backend_type = 'client backend' AND pid <> pg_backend_pid()
GROUP BY state ORDER BY sessions DESC`;

const BLOCKED_SQL = `
SELECT count(*)::int                                                        AS blocked,
       max(EXTRACT(EPOCH FROM (now() - query_start)))                       AS longest_wait_secs,
       (array_agg((pg_blocking_pids(pid))[1]
          ORDER BY now() - query_start DESC))[1]                            AS suspect_pid
FROM pg_stat_activity
WHERE cardinality(pg_blocking_pids(pid)) > 0`;

const LONG_RUNNING_SQL = `
SELECT pid, EXTRACT(EPOCH FROM (now() - query_start)) AS running_secs, query
FROM pg_stat_activity
WHERE state = 'active' AND backend_type = 'client backend'
  AND pid <> pg_backend_pid() AND now() - query_start > interval '5 seconds'
ORDER BY running_secs DESC LIMIT 3`;

const CONNECTIONS_SQL = `
SELECT count(*)::int                                       AS used,
       current_setting('max_connections')::int             AS max
FROM pg_stat_activity`;

const WRAPAROUND_SQL = `
SELECT round(100.0 * max(age(datfrozenxid))
       / current_setting('autovacuum_freeze_max_age')::float)::int AS pct
FROM pg_database`;

const SLOTS_SQL = `
SELECT count(*) FILTER (WHERE NOT active)::int AS inactive,
       coalesce(max(pg_wal_lsn_diff(pg_current_wal_lsn(), restart_lsn))
                FILTER (WHERE NOT active), 0)::bigint AS retained_bytes,
       (array_agg(slot_name ORDER BY pg_wal_lsn_diff(pg_current_wal_lsn(), restart_lsn) DESC)
                FILTER (WHERE NOT active))[1] AS worst_slot
FROM pg_replication_slots`;

async function panel<T>(run: () => Promise<T>): Promise<PanelResult<T>> {
  try {
    return { ok: true, data: await run() };
  } catch (err) {
    return { ok: false, error: (err as Error).message };
  }
}

export async function fetchTriage(): Promise<TriageData> {
  const [sessions, blocked, longRunning, connections, wraparound, slots] = await Promise.all([
    panel(async () => {
      const rows = await query<Record<string, unknown>>(SESSIONS_SQL);
      const byState = rows.map((r) => ({
        state: (r.state as string) ?? 'unknown',
        sessions: Number(r.sessions),
        longestInStateSecs: Number(r.longest_secs ?? 0),
      }));
      const idle = byState.find((s) => s.state.startsWith('idle in transaction'));
      return { byState, oldestIdleInTxnSecs: idle ? idle.longestInStateSecs : null };
    }),
    panel(async () => {
      const [r] = await query<Record<string, unknown>>(BLOCKED_SQL);
      return {
        blockedCount: Number(r?.blocked ?? 0),
        longestWaitSecs: r?.longest_wait_secs == null ? null : Number(r.longest_wait_secs),
        suspectPid: r?.suspect_pid == null ? null : Number(r.suspect_pid),
      };
    }),
    panel(async () => {
      const rows = await query<Record<string, unknown>>(LONG_RUNNING_SQL);
      return rows.map((r) => ({ pid: Number(r.pid), runningSecs: Number(r.running_secs), query: (r.query as string) ?? '' }));
    }),
    panel(async () => {
      const [r] = await query<Record<string, unknown>>(CONNECTIONS_SQL);
      const used = Number(r?.used ?? 0);
      const max = Number(r?.max ?? 0);
      return { used, max, headroom: max - used };
    }),
    panel(async () => {
      const [r] = await query<Record<string, unknown>>(WRAPAROUND_SQL);
      return { pctOfHorizon: Number(r?.pct ?? 0) };
    }),
    panel(async () => {
      const [r] = await query<Record<string, unknown>>(SLOTS_SQL);
      return {
        inactiveCount: Number(r?.inactive ?? 0),
        retainedBytes: Number(r?.retained_bytes ?? 0),
        worstSlot: (r?.worst_slot as string) ?? null,
      };
    }),
  ]);
  return { sessions, blocked, longRunning, connections, wraparound, slots };
}
